cmake_minimum_required(VERSION 3.20)
project(oneapi_sycl_smoke LANGUAGES CXX)
add_executable(sycl-smoke smoke.cpp)
target_compile_features(sycl-smoke PRIVATE cxx_std_17)
target_compile_options(sycl-smoke PRIVATE -fsycl)
target_link_options(sycl-smoke PRIVATE -fsycl)
if(MSVC)
  target_compile_options(sycl-smoke PRIVATE /EHsc)
endif()
